# Standalone tests for the current WIP header; deliberately no x_generate.
CMAKE_MINIMUM_REQUIRED(VERSION 3.18)
PROJECT(x_h_c_tests LANGUAGES C)

ENABLE_TESTING()
FIND_PACKAGE(Python3 COMPONENTS Interpreter REQUIRED)

INCLUDE(${CMAKE_CURRENT_SOURCE_DIR}/modules.cmake)
SET(ERROR_CONTEXT "$<IF:$<OR:$<CONFIG:Release>,$<CONFIG:RelWithDebInfo>,$<CONFIG:MinSizeRel>>,release,debug>")

FOREACH(STANDARD IN ITEMS 99 11)
  SET(TARGET x_test_check_diagnostics_c${STANDARD})
  ADD_EXECUTABLE(${TARGET} check_diagnostics.c)
  ADD_DEPENDENCIES(${TARGET} x_header_minimal)
  TARGET_INCLUDE_DIRECTORIES(${TARGET} PRIVATE
    ${CMAKE_CURRENT_BINARY_DIR}/generated/minimal)
  SET_TARGET_PROPERTIES(${TARGET} PROPERTIES
    C_STANDARD ${STANDARD} C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
  IF(MSVC)
    TARGET_COMPILE_OPTIONS(${TARGET} PRIVATE /W4 /WX)
  ELSE()
    TARGET_COMPILE_OPTIONS(${TARGET} PRIVATE -Wall -Wextra -Werror -pedantic-errors)
  ENDIF()
  ADD_TEST(NAME check_diagnostics_c${STANDARD}
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/../check_error_handling.py
      --executable $<TARGET_FILE:${TARGET}> --case check_minimal
      --expect success --context ${ERROR_CONTEXT})
  SET_TESTS_PROPERTIES(check_diagnostics_c${STANDARD} PROPERTIES
    TIMEOUT 15 LABELS "error;modularization")
ENDFOREACH()

SET(ERROR_CASES
  values registration capture messages socket check check_success logging
  assert_success
  capacity cross_tu
)
IF(WIN32)
  LIST(APPEND ERROR_CASES win32)
ENDIF()
IF(APPLE)
  LIST(APPEND ERROR_CASES mach)
ENDIF()
SET(ERROR_ABORT_CASES
  abort_unknown_value abort_unknown_capture abort_null_name abort_empty_name
  abort_unsupported_capture abort_unsupported_custom_capture
  abort_omitted_assertion abort_null_assertion abort_empty_assertion
  abort_message_assertion
)

FOREACH(LOG_MODE IN ITEMS enabled disabled)
  SET(TARGET x_test_error_${LOG_MODE})
  ADD_EXECUTABLE(${TARGET} error_handling.c error_handling_peer.c)
  # Both translation units exercise the retained built-in socket error domain.
  TARGET_COMPILE_DEFINITIONS(${TARGET} PRIVATE X_ENABLE_SOCKET=1)
  SET_TARGET_PROPERTIES(${TARGET} PROPERTIES
    C_STANDARD 99 C_STANDARD_REQUIRED YES C_EXTENSIONS NO
  )
  IF(LOG_MODE STREQUAL "enabled")
    TARGET_COMPILE_DEFINITIONS(${TARGET} PRIVATE X_LOG_LEVEL=2)
  ELSE()
    TARGET_COMPILE_DEFINITIONS(${TARGET} PRIVATE X_LOG_LEVEL=-1)
  ENDIF()
  IF(MSVC)
    TARGET_COMPILE_OPTIONS(${TARGET} PRIVATE /W4)
  ELSE()
    # Do not promote the WIP header's unrelated unused-parameter warnings to
    # errors. Review issue 2 is intentionally not tested with -pedantic-errors.
    TARGET_COMPILE_OPTIONS(${TARGET} PRIVATE -Wall -Wextra)
  ENDIF()
  IF(WIN32)
    # The retained socket error domain captures WSAGetLastError().
    TARGET_LINK_LIBRARIES(${TARGET} PRIVATE ws2_32)
  ELSEIF(UNIX)
    TARGET_LINK_LIBRARIES(${TARGET} PRIVATE m)
  ENDIF()

  # Only logging and check diagnostics need both verbosity configurations.
  IF(LOG_MODE STREQUAL "enabled")
    SET(CASES ${ERROR_CASES} ${ERROR_ABORT_CASES})
  ELSE()
    SET(CASES check logging)
  ENDIF()
  FOREACH(CASE IN LISTS CASES)
    IF(CASE IN_LIST ERROR_ABORT_CASES)
      SET(EXPECT abort)
    ELSE()
      SET(EXPECT success)
    ENDIF()
    IF(CASE STREQUAL "check" OR CASE STREQUAL "logging")
      SET(TEST_NAME error_${LOG_MODE}_${CASE})
    ELSE()
      SET(TEST_NAME error_${CASE})
    ENDIF()
    ADD_TEST(NAME ${TEST_NAME}
      COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/../check_error_handling.py
        --executable $<TARGET_FILE:${TARGET}> --case ${CASE}
        --expect ${EXPECT} --context ${ERROR_CONTEXT}
    )
    SET_TESTS_PROPERTIES(${TEST_NAME} PROPERTIES TIMEOUT 15 LABELS error)
  ENDFOREACH()
ENDFOREACH()
